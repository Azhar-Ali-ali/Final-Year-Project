const express = require('express');
const cors = require('cors');
const morgan = require('morgan');
const dotenv = require('dotenv');
const path = require('path');
const { query, testConnection, closePool } = require('../../database/postgresClient');

const dashboardRoutes = require('./routes/dashboardRoutes');
const cmsRoutes = require('./routes/cmsRoutes');
const disputeRoutes = require('./routes/disputeRoutes');
const logisticsRoutes = require('./routes/logisticsRoutes');
const orderRoutes = require('./routes/orderRoutes');
const paymentPayoutRoutes = require('./routes/paymentPayoutRoutes');
const productCatalogRoutes = require('./routes/productCatalogRoutes');
const reportsAnalyticsRoutes = require('./routes/reportsAnalyticsRoutes');
const reviewsRatingsRoutes = require('./routes/reviewsRatingsRoutes');
const sellerManagementRoutes = require('./routes/sellerManagementRoutes');
const userManagementRoutes = require('./routes/userManagementRoutes');
const adminAuthRoutes = require('./routes/adminAuthRoutes');
const supportRoutes = require('./routes/supportRoutes');
const returnsRoutes = require('./routes/returnsRoutes');

dotenv.config();

const app = express();
const PORT = Number(process.env.PORT) || 5000;
const FRONTEND_ORIGIN = process.env.FRONTEND_ORIGIN || '*';

app.use(cors({ origin: FRONTEND_ORIGIN === '*' ? true : FRONTEND_ORIGIN }));
app.use(express.json({ limit: '50mb' }));
app.use(express.urlencoded({ extended: true, limit: '50mb' }));
app.use(morgan('dev'));

app.locals.db = { query };

app.use((req, res, next) => {
  req.db = app.locals.db;
  next();
});

app.get('/api/health', async (req, res) => {
  try {
    const dbInfo = await testConnection();
    res.json({
      success: true,
      message: 'Admin backend is running',
      database: {
        connected: true,
        name: dbInfo.db,
        serverTime: dbInfo.serverTime
      }
    });
  } catch (error) {
    res.status(503).json({
      success: false,
      message: 'Admin backend is running but database is unavailable',
      database: {
        connected: false,
        error: error.message || error.code || 'Database connection failed'
      }
    });
  }
});

app.use('/api/admin/dashboard', dashboardRoutes);
app.use('/api/admin/cms', cmsRoutes);
app.use('/api/admin/disputes', disputeRoutes);
app.use('/api/admin/logistics', logisticsRoutes);
app.use('/api/admin/orders', orderRoutes);
app.use('/api/admin/payments', paymentPayoutRoutes);
app.use('/api/admin/catalog', productCatalogRoutes);
app.use('/api/admin/reports', reportsAnalyticsRoutes);
app.use('/api/admin/reviews', reviewsRatingsRoutes);
app.use('/api/admin/sellers', sellerManagementRoutes);
app.use('/api/admin/users', userManagementRoutes);
app.use('/api/admin/auth', adminAuthRoutes);
app.use('/api/admin/support', supportRoutes);
app.use('/api/admin/returns', returnsRoutes);

const adminPagesPath = path.resolve(__dirname, '../../..', 'Front_End/Admin/pages');
const adminJsPath = path.resolve(__dirname, '../../..', 'Front_End/Admin/js');
app.use('/admin-pages', express.static(adminPagesPath));
app.use('/js', express.static(adminJsPath));

app.use((req, res) => {
  res.status(404).json({ success: false, message: 'Route not found' });
});

async function startServer() {
  try {
    const dbInfo = await testConnection();
    console.log(`[Admin API] Database connected: ${dbInfo.db}`);
  } catch (error) {
    console.warn(`[Admin API] Database connection failed: ${error.message}`);
  }

  app.listen(PORT, () => {
    console.log(`Admin backend running on http://localhost:${PORT}`);
  });
}

process.on('SIGINT', async () => {
  await closePool();
  process.exit(0);
});

process.on('SIGTERM', async () => {
  await closePool();
  process.exit(0);
});

startServer();
